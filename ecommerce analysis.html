import datetime as dt

# Analysis reference date - dataset-la last date-ku 1 day appuram
reference_date = df['InvoiceDate'].max() + dt.timedelta(days=1)

# RFM calculate pannurom
rfm = df.groupby('CustomerID').agg({
    'InvoiceDate': lambda x: (reference_date - x.max()).days,   # Recency
    'InvoiceNo': 'nunique',                                      # Frequency
    'TotalPrice': 'sum'                                          # Monetary
})

rfm.columns = ['Recency', 'Frequency', 'Monetary']

# Ovoru RFM value-kum score kudukurom (1 to 4, 4 = best)
rfm['R_Score'] = pd.qcut(rfm['Recency'], 4, labels=[4,3,2,1])
rfm['F_Score'] = pd.qcut(rfm['Frequency'].rank(method='first'), 4, labels=[1,2,3,4])
rfm['M_Score'] = pd.qcut(rfm['Monetary'], 4, labels=[1,2,3,4])

# 3 scores-ah combine panni oru RFM Score create pannurom
rfm['RFM_Score'] = rfm['R_Score'].astype(str) + rfm['F_Score'].astype(str) + rfm['M_Score'].astype(str)

# RFM score total calculate pannurom (Recency + Frequency + Monetary scores)
rfm['RFM_Total'] = rfm[['R_Score','F_Score','M_Score']].astype(int).sum(axis=1)

# Total score vachu customer segment name kudukurom
def segment_customer(score):
    if score >= 10:
        return 'Best Customers'
    elif score >= 7:
        return 'Loyal Customers'
    elif score >= 5:
        return 'At Risk'
    else:
        return 'Lost Customers'

rfm['Segment'] = rfm['RFM_Total'].apply(segment_customer)

# Ovoru segment-la evlo customers irukku nu count pannurom
print(rfm['Segment'].value_counts())

# Chart-ah kaatalam
plt.figure(figsize=(8,5))
rfm['Segment'].value_counts().plot(kind='bar', color='purple')
plt.title('Customer Segments')
plt.xlabel('Segment')
plt.ylabel('Number of Customers')
plt.tight_layout()
plt.show()
